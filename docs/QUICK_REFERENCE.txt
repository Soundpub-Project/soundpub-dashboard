┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

  -- Email verification rate (7 days)
  SELECT 
    COUNT(*) FILTER (WHERE email_verified = true) * 100.0 / COUNT(*) AS rate
  FROM soundpub.profiles
  WHERE created_at >= NOW() - INTERVAL '7 days';

  -- Password reset requests (today)
  SELECT COUNT(*) FROM soundpub.auth_events
  WHERE event_type = 'password_reset_requested'
    AND created_at >= CURRENT_DATE;

  -- Rate limit violations
  SELECT COUNT(*) FROM soundpub.rate_limits
  WHERE blocked_until > NOW();

  -- Cleanup functions
  SELECT soundpub.cleanup_rate_limits();
  SELECT soundpub.cleanup_expired_tokens();
  SELECT soundpub.cleanup_old_auth_events();

  RATE LIMIT STUCK:
    → Reset: DELETE FROM soundpub.rate_limits 
             WHERE identifier = 'user@example.com';

  USER UNVERIFIED:
    → Fix: UPDATE soundpub.profiles SET email_verified = true
